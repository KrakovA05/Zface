    ```js
    const url = Linking.useLinkingURL();
    useEffect(() => {
      if (!url || !url.includes('access_token')) return;
      const { params, errorCode } = QueryParams.getQueryParams(url);
      if (errorCode) {
        setDeepLinkStatus('error');
        setDeepLinkError(errorCode);
        return;
      }
      const { access_token, refresh_token } = params;
      if (!access_token) return;
      supabase.auth.setSession({ access_token, refresh_token })
        .then(async ({ data, error }) => {
          if (error || !data.session) {
            setDeepLinkStatus('error');
            setDeepLinkError(error?.message || 'session_error');
            return;
          }
          await supabase.from('users').upsert({
            user_id: data.user.id,
            username,
            email,
            labels: labels || [],
          }, { onConflict: 'user_id' });
          store.userId = data.user.id;
          store.username = username;
          store.email = email;
          setDeepLinkStatus('success');
          setTimeout(() => navigation.navigate('Test'), 1500);
        });
    }, [url]);
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| noy://...?access_token=... → App | Токены из URL попадают в supabase.auth.setSession() |
| Supabase → noy:// redirect | Supabase отправляет токены на deep link URL |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-03-01 | Spoofing | Deep link token interception | mitigate | HTTPS redirect URL (Supabase использует HTTPS), PKCE flow включён в supabase-js v2 по умолчанию. Redirect URL ограничен в Dashboard только noy://** |
| T-07-03-02 | Elevation of Privilege | Open redirect в Supabase | mitigate | В Additional Redirect URLs разрешить только "noy://**" — никаких wildcard на другие схемы или HTTP URLs |
| T-07-03-03 | Spoofing | Поддельный deep link | accept | Другое приложение не может зарегистрировать ту же схему noy:// на iOS/Android без подписи — система блокирует |
| T-07-03-04 | Information Disclosure | access_token в URL | accept | Supabase JS v2 с PKCE использует короткоживущие токены. setSession() немедленно валидирует токен через Supabase API. |
</threat_model>
